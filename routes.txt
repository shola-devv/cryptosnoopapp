## How It Works

1. **`[account]` and `[address]`** are **dynamic route segments**
2. They capture the ID from the URL
3. Access them in your code with `context.params.account` or `context.params.address`

// Example: Create Asset Form
'use client';
import { usePortfolio } from '../hooks/usePortfolio';

export function CreateAssetForm() {
  const { refreshAssets } = usePortfolio();

  const handleSubmit = async (formData) => {
    const response = await fetch('/api/assets', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(formData),
    });

    if (response.ok) {
      // Immediately refresh assets
      refreshAssets();
    }
  };

  return <form onSubmit={handleSubmit}>{/* form fields */}</form>;
}
